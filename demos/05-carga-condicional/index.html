/**
 * Patrón Senior: Guard no-bloqueante con límite de seguridad (Fail-Safe)
 * Erradica setTimeout(fn, 300) y previene bloqueos por 404
 */
function initWhenReady(maxWaitFrames = 240) {
  let frameCount = 0;

  function check() {
    frameCount++;

    // 1. Verificación en runtime del módulo
    if (typeof window.gsap !== 'undefined') {
      return executeModule(window.gsap);
    }

    // 2. Fail-Safe: Abortar si supera el umbral de espera (~4 segundos)
    if (frameCount > maxWaitFrames) {
      console.warn('[Guard] Dependencia no disponible tras timeout. Degradación segura.');
      return renderFallbackState();
    }

    // 3. Reintentar en el siguiente ciclo de refresco
    requestAnimationFrame(check);
  }

  requestAnimationFrame(check);
}
